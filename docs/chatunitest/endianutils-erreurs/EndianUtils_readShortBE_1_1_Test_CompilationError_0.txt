class EndianUtils_readShortBE_1_1_Test {

    @Test
    void testReadShortBE() throws IOException, BufferUnderrunException, NoSuchMethodException, IllegalAccessException, InvocationTargetException {
        // Create a ByteArrayInputStream with a short value in big-endian format
        byte[] bytes = new byte[2];
        bytes[0] = (byte) 0x12;
        bytes[1] = (byte) 0x34;
        ByteArrayInputStream stream = new ByteArrayInputStream(bytes);
        // Create an instance of EndianUtils using reflection
        EndianUtils endianUtils = new EndianUtils();
        // Get the private method readUShortBE using reflection
        Method readUShortBE = EndianUtils.class.getDeclaredMethod("readUShortBE", InputStream.class);
        readUShortBE.setAccessible(true);
        // Invoke the private method readUShortBE using reflection
        int result = (int) readUShortBE.invoke(endianUtils, stream);
        // Check that the result is equal to the expected value
        assertEquals(0x1234, result);
        // Check that the stream is not closed after invoking the method
        assertFalse(stream.available() == 0);
    }
}
