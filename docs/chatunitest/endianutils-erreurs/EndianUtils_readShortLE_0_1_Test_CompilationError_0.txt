package org.apache.tika.io;

import org.apache.tika.io.EndianUtils;
import org.apache.tika.io.BufferUnderrunException;
import java.io.ByteArrayInputStream;
import java.io.IOException;
import java.io.InputStream;
import org.mockito.*;
import org.junit.jupiter.api.*;
import static org.mockito.Mockito.*;
import static org.junit.jupiter.api.Assertions.*;
import org.junit.jupiter.api.extension.ExtendWith;
import org.mockito.junit.jupiter.MockitoExtension;
import org.apache.tika.exception.TikaException;

public class EndianUtils_readShortLE_0_1_Test {

    @Test
    public void testReadShortLE() throws IOException, BufferUnderrunException {
        // Test with a valid short value
        InputStream stream = new ByteArrayInputStream(new byte[] { (byte) 0x12, (byte) 0x34 });
        short result = EndianUtils.readShortLE(stream);
        assertEquals(0x3412, result);
        // Test with a negative short value
        stream = new ByteArrayInputStream(new byte[] { (byte) 0xFF, (byte) 0xFF });
        result = EndianUtils.readShortLE(stream);
        assertEquals(-1, result);
        // Test with a buffer underrun exception
        stream = new ByteArrayInputStream(new byte[] {});
        assertThrows(BufferUnderrunException.class, () -> {
            EndianUtils.readShortLE(stream);
        });
    }
}
