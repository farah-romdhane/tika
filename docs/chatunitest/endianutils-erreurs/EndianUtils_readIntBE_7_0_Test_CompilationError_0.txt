package org.apache.tika.io;

import java.io.ByteArrayInputStream;
import java.io.IOException;
import java.io.InputStream;
import org.apache.tika.exception.TikaException;
import org.apache.tika.io.EndianUtils;
import org.apache.tika.io.BufferUnderrunException;
import org.mockito.*;
import org.junit.jupiter.api.*;
import static org.mockito.Mockito.*;
import static org.junit.jupiter.api.Assertions.*;
import org.junit.jupiter.api.extension.ExtendWith;
import org.mockito.junit.jupiter.MockitoExtension;

public class EndianUtils_readIntBE_7_0_Test {

    private InputStream mockInputStream;

    @BeforeEach
    public void setUp() {
        mockInputStream = mock(InputStream.class);
    }

    @Test
    public void testReadIntBE() throws IOException, TikaException {
        // Arrange
        when(mockInputStream.read()).thenReturn(0x12, 0x34, 0x56, 0x78);
        // Act
        int result = EndianUtils.readIntBE(mockInputStream);
        // Assert
        assertEquals(0x12345678, result);
    }

    @Test
    public void testReadIntBE_WithBufferUnderrun() throws IOException, TikaException {
        // Arrange
        when(mockInputStream.read()).thenReturn(0x12, 0x34, 0x56, -1);
        // Act & Assert
        assertThrows(BufferUnderrunException.class, () -> EndianUtils.readIntBE(mockInputStream));
    }
}
